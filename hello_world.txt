    <div class="container">
        <h1>Sudoku</h1>

            <button onclick="newGame()">New Game</button>

    <script>
        let board = [];
        let solution = [];
        let initialBoard = [];
        let selectedCell = null;
        let timerInterval = null;
        let secondsElapsed = 0;

        // Simple pre-defined valid Sudoku templates (Puzzle + Solution)
        const sampleGames = [
            {
                puzzle: [
                    5,3,0, 0,7,0, 0,0,0,
                    6,0,0, 1,9,5, 0,0,0,
                    0,9,8, 0,0,0, 0,6,0,
                    8,0,0, 0,6,0, 0,0,3,
                    4,0,0, 8,0,3, 0,0,1,
                    7,0,0, 0,2,0, 0,0,6,
                    0,6,0, 0,0,0, 2,8,0,
                    0,0,0, 4,1,9, 0,0,5,
                    0,0,0, 0,8,0, 0,7,9
                ],
                solution: [
                    5,3,4, 6,7,8, 9,1,2,
                    6,7,2, 1,9,5, 3,4,8,
                    1,9,8, 3,4,2, 5,6,7,
                    8,5,9, 7,6,1, 4,2,3,
                    4,2,6, 8,5,3, 7,9,1,
                    7,1,3, 9,2,4, 8,5,6,
                    9,6,1, 5,3,7, 2,8,4,
                    2,8,7, 4,1,9, 6,3,5,
                    3,4,5, 2,8,6, 1,7,9
                ]
            }
        ];

        function initGame() {
            const game = sampleGames[0];
            initialBoard = [...game.puzzle];
            board = [...game.puzzle];
            solution = [...game.solution];

            renderBoard();
            startTimer();
            document.getElementById('win-message').style.display = 'none';
        }

        function renderBoard() {
            const boardContainer = document.getElementById('sudoku-board');
            boardContainer.innerHTML = '';

            for (let i = 0; i < 81; i++) {
                const cell = document.createElement('div');
                cell.classList.add('cell');
                cell.dataset.index = i;

                if (board[i] !== 0) {
                    cell.textContent = board[i];
                    if (initialBoard[i] !== 0) {
                        cell.classList.add('fixed');
                    }
                }

                cell.addEventListener('click', () => selectCell(cell, i));
                boardContainer.appendChild(cell);
            }
        }

        function selectCell(cellElement, index) {
            if (initialBoard[index] !== 0) return; // Can't edit fixed clues

            document.querySelectorAll('.cell').forEach(c => c.classList.remove('selected'));
            cellElement.classList.add('selected');
            selectedCell = index;
        }

        // Handle keyboard input for numbers (1-9) and deletion (Backspace/Delete)
        document.addEventListener('keydown', (e) => {
            if (selectedCell === null) return;

            const num = parseInt(e.key);
            if (num >= 1 && num <= 9) {
                board[selectedCell] = num;
                updateCellView(selectedCell, num);
            } else if (e.key === 'Backspace' || e.key === 'Delete') {
                board[selectedCell] = 0;
                updateCellView(selectedCell, '');
            }
        });

        function updateCellView(index, value) {
            const cells = document.querySelectorAll('.cell');
            cells[index].textContent = value === 0 ? '' : value;
            cells[index].classList.remove('error');
        }

        function checkSolution() {
            let isComplete = true;
            let isCorrect = true;
            const cells = document.querySelectorAll('.cell');

            for (let i = 0; i < 81; i++) {
                cells[i].classList.remove('error');
                if (board[i] === 0) {
                    isComplete = false;
                } else if (board[i] !== solution[i]) {
                    isCorrect = false;
                    cells[i].classList.add('error');
                }
            }

            const winMessage = document.getElementById('win-message');
            if (isComplete && isCorrect) {
                winMessage.style.display = 'block';
                clearInterval(timerInterval);
            } else if (!isCorrect) {
                alert("There are some errors highlighted in red. Keep trying!");
            } else if (!isComplete) {
                alert("The board is not completely filled yet!");
            }
        }

        function startTimer() {
            clearInterval(timerInterval);
            secondsElapsed = 0;
            timerInterval = setInterval(() => {
                secondsElapsed++;
                const mins = String(Math.floor(secondsElapsed / 60)).padStart(2, '0');
                const secs = String(secondsElapsed % 60).padStart(2, '0');
                document.getElementById('timer').textContent = `Time: ${mins}:${secs}`;
            }, 1000);
        }

        function newGame() {
            initGame();
        }

        // Initialize on load
        initGame();
    </script>
